### app/page.tsx

```tsx
import { useEffect, useState } from "react";
import { useAuth } from "@clerk/nextjs";
import TaskList from "@/components/task-list";
import AddTaskDialog from "@/components/add-task-dialog";

export default function Home() {
  const { userId } = useAuth();
  const [tasks, setTasks] = useState<Task[]>([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<ApiErrorResponse | null>(null);

  useEffect(() => {
    if (!userId) return;

    const fetchTasks = async () => {
      try {
        const response = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/api/v1/tasks`, {
          headers: {
            Authorization: `Bearer ${await auth().getToken()}`,
          },
        });

        if (!response.ok) {
          setError({ status: response.status, message: "Failed to fetch tasks" });
          return;
        }

        const data = (await response.json()) as PaginatedResponse<Task>;
        setTasks(data.items);
      } catch (err) {
        setError({ status: 500, message: "Internal server error" });
      } finally {
        setIsLoading(false);
      }
    };

    fetchTasks();
  }, [userId]);

  return (
    <div className="flex flex-col items-center p-4">
      {error && (
        <div className="text-red-600 mt-4">{error.message}</div>
      )}
      <AddTaskDialog />
      <h1 className="mt-8 text-2xl font-bold">Tasks</h1>
      {isLoading ? (
        <div>Loading...</div>
      ) : tasks.length === 0 ? (
        <div>No tasks found.</div>
      ) : (
        <TaskList tasks={tasks} />
      )}
    </div>
  );
}
```

### app/(auth)/login/page.tsx

```tsx
import { useAuth } from "@clerk/nextjs";
import { useRouter } from "next/navigation";

export default function LoginPage() {
  const { signOut, isSignedIn } = useAuth();
  const router = useRouter();

  useEffect(() => {
    if (isSignedIn) {
      router.push("/");
    }
  }, [isSignedIn, router]);

  return (
    <div className="flex flex-col items-center justify-center h-screen">
      <h1>Login</h1>
      <button onClick={() => signOut()} className="mt-4 px-4 py-2 bg-gray-300 text-black rounded">
        Sign In
      </button>
    </div>
  );
}
```
