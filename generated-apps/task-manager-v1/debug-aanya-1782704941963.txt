```tsx
import { useEffect, useState } from 'react';
import { ClerkProvider, useUser, SignedIn, SignedOut } from '@clerk/nextjs';
import { TaskStatus } from '@/shared-types';
import StatusFilter from '@/components/StatusFilter';
import TaskItem from '@/components/TaskItem';

const fetchTasks = async (status?: TaskStatus, page = 1, perPage = 10) => {
  const url = new URL(`${process.env.NEXT_PUBLIC_API_URL}/api/v1/tasks`, window.location.origin);
  url.searchParams.append('page', String(page));
  url.searchParams.append('perPage', String(perPage));
  if (status) url.searchParams.append('status', status);

  const res = await fetch(url, {
    headers: {
      'Authorization': `Bearer ${await auth().getToken()}`,
    },
  });

  if (!res.ok) {
    throw new Error((await res.json())?.message || 'Failed to fetch tasks');
  }

  return (await res.json()) as PaginatedResponse<Task>;
};

const TasksPage = () => {
  const { user } = useUser();
  const [tasks, setTasks] = useState<Task[]>([]);
  const [statusFilter, setStatusFilter] = useState<TaskStatus | null>(null);

  useEffect(() => {
    if (!user) return;

    const fetchTaskData = async () => {
      try {
        const data = await fetchTasks(statusFilter);
        setTasks(data.items);
      } catch (error) {
        console.error(error);
      }
    };

    fetchTaskData();
  }, [statusFilter, user]);

  return (
    <SignedIn>
      <div className="p-4">
        <StatusFilter onStatusChange={(status) => setStatusFilter(status)} />
        <ul className="mt-4 space-y-2">
          {tasks.map((task) => (
            <TaskItem key={task.id} task={task} />
          ))}
        </ul>
      </div>
    </SignedIn>
  );
};

export default TasksPage;
```

===FILE: components/TaskForm.tsx===
```tsx
import { useForm, SubmitHandler } from 'react-hook-form';
import { useAuth } from '@clerk/nextjs';
import { Task, CreateTaskRequest, UpdateTaskRequest } from '@/shared-types';

interface TaskFormProps {
  task?: Task;
  onSubmit: (task: Task) => void;
}

const TaskForm = ({ task = null, onSubmit }: TaskFormProps) => {
  const { register, handleSubmit, reset, setValue } = useForm<CreateTaskRequest | UpdateTaskRequest>({
    defaultValues: {
      title: task?.title || '',
      description: task?.description || '',
      dueDate: task?.dueDate ? task.dueDate.toISOString().split('T')[0] : undefined,
    },
  });

  useEffect(() => {
    if (task) {
      setValue('title', task.title);
      setValue('description', task.description || '');
      setValue('dueDate', task.dueDate ? task.dueDate.toISOString().split('T')[0] : undefined);
    } else {
      reset();
    }
  }, [reset, setValue, task]);

  const { userId } = useAuth();

  const onSubmitHandler: SubmitHandler<CreateTaskRequest | UpdateTaskRequest> = async (data) => {
    try {
      let url = `${process.env.NEXT_PUBLIC_API_URL}/api/v1/tasks`;
      if (task) {
        url += `/${task.id}`;
        data = { ...data, isCompleted: task.isCompleted };
      } else {
        data = { ...data, userId };
      }

      const res = await fetch(url, {
        method: task ? 'PUT' : 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${await auth().getToken()}`,
        },
        body: JSON.stringify(data),
      });

      if (!res.ok) {
        throw new Error((await res.json())?.message || 'Failed to submit task');
      }

      onSubmit(await res.json());
    } catch (error) {
      console.error(error);
    }
  };

  return (
    <form onSubmit={handleSubmit(onSubmitHandler)}>
      <div className="mb-4">
        <label htmlFor="title" className="block text-sm font-medium text-gray-700 dark:text-white">
          Title
        </label>
        <input
          type="text"
          id="title"
          {...register('title', { required: 'Title is required' })}
          className="mt-1 block w-full border-gray-300 rounded-md shadow-sm focus:border-indigo-500 focus:ring-indigo-500 sm:text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
      </div>
      <div className="mb-4">
        <label htmlFor="description" className="block text-sm font-medium text-gray-700 dark:text-white">
          Description
        </label>
        <textarea
          id="description"
          {...register('description')}
          className="mt-1 block w-full border-gray-300 rounded-md shadow-sm focus:border-indigo-500 focus:ring-indigo-500 sm:text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
      </div>
      <div className="mb-4">
        <label htmlFor="dueDate" className="block text-sm font-medium text-gray-700 dark:text-white">
          Due Date
        </label>
        <input
          type="date"
          id="dueDate"
          {...register('dueDate')}
          className="mt-1 block w-full border-gray-300 rounded-md shadow-sm focus:border-indigo-500 focus:ring-indigo-500 sm:text-sm dark:border-gray-600 dark:bg-gray-700 dark:text-white"
        />
      </div>
      <button type="submit" className="bg-indigo-600 text-white px-4 py-2 rounded-md hover:bg-indigo-700">
        Submit
      </button>
    </form>
  );
};

export default TaskForm;
```

```tsx
import { useState } from 'react';

interface StatusFilterProps {
  onStatusChange: (status: TaskStatus | null) => void;
}

const StatusFilter = ({ onStatusChange }: StatusFilterProps) => {
  const [selectedStatus, setSelectedStatus] = useState<TaskStatus | null>(null);

  return (
    <div className="flex space-x-4">
      <button
        onClick={() => {
          setSelectedStatus(null);
          onStatusChange(null);
        }}
        className={`px-2 py-1 rounded-md border ${
          selectedStatus === null ? 'bg-indigo-600 text-white' : 'border-gray-300 text-gray-700'
        }`}
      >
        All
      </button>
      <button
        onClick={() => {
          setSelectedStatus(TaskStatus.PENDING);
          onStatusChange(TaskStatus.PENDING);
        }}
        className={`px-2 py-1 rounded-md border ${
          selectedStatus === TaskStatus.PENDING ? 'bg-indigo-600 text-white' : 'border-gray-300 text-gray-700'
        }`}
      >
        Pending
      </button>
      <button
        onClick={() => {
          setSelectedStatus(TaskStatus.COMPLETED);
          onStatusChange(TaskStatus.COMPLETED);
        }}
        className={`px-2 py-1 rounded-md border ${
          selectedStatus === TaskStatus.COMPLETED ? 'bg-indigo-600 text-white' : 'border-gray-300 text-gray-700'
        }`}
      >
        Completed
      </button>
    </div>
  );
};

export default StatusFilter;
```
