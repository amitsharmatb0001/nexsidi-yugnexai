```tsx
"use client";

import { useEffect, useState } from "react";
import { useRouter } from "next/navigation";
import { useAuth } from "@clerk/nextjs";
import { TaskStatus, PaginatedResponse } from "@/shared/types";
import { fetchTasks } from "@/lib/api";
import TasksList from "@/components/TasksList";

const Home = () => {
  const router = useRouter();
  const { getToken, isLoaded, isSignedIn } = useAuth();
  const [tasks, setTasks] = useState<Task[]>([]);
  const [page, setPage] = useState(1);
  const [perPage, setPerPage] = useState(20);

  useEffect(() => {
    if (!isLoaded) return;
    if (!isSignedIn) router.push("/sign-in");
    fetchTasks(page, perPage).then((data) => setTasks(data.items));
  }, [isLoaded, isSignedIn, page, perPage, router]);

  const handlePagination = (newPage: number) => {
    setPage(newPage);
  };

  return (
    <div className="container mx-auto p-4">
      <h1 className="text-3xl font-bold mb-8">Tasks</h1>
      <TasksList
        tasks={tasks}
        onPaginationChange={handlePagination}
        currentPage={page}
        perPage={perPage}
      />
    </div>
  );
};

export default Home;
```

```typescript
import { useAuth } from "@clerk/nextjs";
import { PaginatedResponse, TaskStatus } from "@/shared/types";

const fetchTasks = async (page: number = 1, perPage: number = 20, status?: TaskStatus): Promise<PaginatedResponse<Task>> => {
  const { getToken } = useAuth();
  const token = await getToken();
  const apiUrl = process.env.NEXT_PUBLIC_API_URL || "http://localhost:3001";
  const url = new URL(apiUrl + "/api/v1/tasks");
  if (status) url.searchParams.set("status", status);
  url.searchParams.set("page", page.toString());
  url.searchParams.set("perPage", perPage.toString());

  const res = await fetch(url, {
    method: "GET",
    headers: { Authorization: "Bearer " + token },
  });

  if (!res.ok) {
    throw new Error(`HTTP error! status: ${res.status}`);
  }

  return res.json();
};

export default fetchTasks;
```

```tsx
import { Task, TaskStatus } from "@/shared/types";

interface TasksListProps {
  tasks: Task[];
  onPaginationChange: (newPage: number) => void;
  currentPage: number;
  perPage: number;
}

const TasksList = ({ tasks, onPaginationChange, currentPage, perPage }: TasksListProps) => {
  return (
    <div>
      <ul className="divide-y divide-gray-200">
        {tasks.map((task) => (
          <li key={task.id} className="py-4">
            <div className="flex justify-between items-center">
              <span>{task.title}</span>
              <span>{task.isCompleted ? "Completed" : "Pending"}</span>
            </div>
          </li>
        ))}
      </ul>
      <div className="mt-4 flex justify-center space-x-2">
        <button
          onClick={() => onPaginationChange(currentPage - 1)}
          disabled={currentPage === 1}
          className="bg-blue-500 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded focus:outline-none focus:shadow-outline"
        >
          Previous
        </button>
        <button
          onClick={() => onPaginationChange(currentPage + 1)}
          disabled={currentPage === Math.ceil(tasks.length / perPage)}
          className="bg-blue-500 hover:bg-blue-700 text-white font-bold py-2 px-4 rounded focus:outline-none focus:shadow-outline"
        >
          Next
        </button>
      </div>
    </div>
  );
};

export default TasksList;
```

===FILE: app/api/tasks.ts===
```typescript
import { NextRequest, NextResponse } from "next/server";
import { TaskStatus, PaginatedResponse } from "@/shared/types";
import dbConnect from "@/lib/dbConnect";
import TaskModel from "@/models/Task";

const getTasks = async (req: NextRequest) => {
  const { searchParams } = new URL(req.url);
  const page = parseInt(searchParams.get("page") || "1");
  const perPage = parseInt(searchParams.get("perPage") || "20");
  const status = searchParams.get("status") as TaskStatus | undefined;

  await dbConnect();

  const query: any = {};
  if (status) {
    query.status = status;
  }

  const tasks = await TaskModel.find(query)
    .sort({ createdAt: -1 })
    .skip((page - 1) * perPage)
    .limit(perPage);

  const count = await TaskModel.countDocuments(query);
  const totalPages = Math.ceil(count / perPage);

  return NextResponse.json({
    data: tasks,
    pagination: {
      currentPage,
      perPage,
      totalPages,
      totalItems: count,
    },
  });
};

export { getTasks };
```

```typescript
import mongoose from "mongoose";

let MONGODB_URI = process.env.MONGODB_URI;

if (!MONGODB_URI) {
  throw new Error(
    'Please define the MONGODB_URI environment variable inside .env.local'
  );
}

const opts = {
  useNewUrlParser: true,
  useUnifiedTopology: true,
};

async function dbConnect() {
  if (mongoose.connections[0].readyState === 1) {
    return;
  }

  try {
    await mongoose.connect(MONGODB_URI, opts);
    console.log("MongoDB connected");
  } catch (error) {
    console.error(error);
  }
}

export default dbConnect;
```
